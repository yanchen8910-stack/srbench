
Building code:
for ALG in algorithms/*/; do
    SUBNAME=$(basename "$ALG")
    if [ "$SUBNAME" == "afp" ]; then continue; fi
    echo "========================================"
    echo "Building: $SUBNAME"
    echo "========================================"
    docker build -t srbench/${SUBNAME} -f alg-Dockerfile --build-arg ALGORITHM=${SUBNAME} . 2>&1 | tail -5
    if [ $? -ne 0 ]; then
        echo "FAILED: $SUBNAME"
    fi
done
